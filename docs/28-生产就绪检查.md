```bash
cd cad && ../.venv/bin/python build.py && \
../.venv/bin/python manufacture.py && \
../.venv/bin/python verify_design.py && \
../.venv/bin/python cost.py && \
../.venv/bin/python harness.py && \
../.venv/bin/python design_space.py && \
../.venv/bin/python parts_index.py && \
cd ../firmware && make test && \
cd ../software && PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 ../.venv/bin/python -m pytest -q
```
